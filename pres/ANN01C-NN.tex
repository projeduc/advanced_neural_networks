\section{Neuron}

\begin{frame}
	\frametitle{\insertshortsubtitle}
	\framesubtitle{\insertsection}
	
	\begin{itemize}
		\item Let us recall \optword{Logistic regression}
		\item Features are combined linearly: this can be seen as a accumulation of signals
		\[z = \theta_0 + \sum\limits_{j=1}^{N} \theta_j x_j\]
		\item Logistic function is applied on this sum: It is called \keyword{activation function} (it approximates a thresholding behavior observed in biological neurons)
		\[\sigma(z) = \frac{1}{1 + e^{-z}}\] 
		\item To measure the prediction error, a \keyword{cost function} is used
		\item To train the parameters, an \keyword{optimization algorithm} is used
	\end{itemize}

\end{frame}

\subsection{Activation functions}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item A neuron first computes a linear combination of inputs:
		\[
		z = \theta_0 + \sum\limits_{j=1}^{N} \theta_j x_j
		\]
		
		\item If we only use linear transformations, stacking multiple layers is equivalent to a single linear model
		
		\item \keyword{Activation functions} introduce non-linearity into the model
		
		\item This allows neural networks to approximate complex, non-linear relationships
		
		\item Some activation functions can be interpreted as smooth approximations of thresholding behavior
	\end{itemize}
	
\end{frame}

\subsection{Input layer}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item The input layer receives the feature vector:
		\[
		x = (x_1, x_2, \dots, x_N)
		\]
		
		\item Each neuron corresponds to one input feature
		
		\item No transformation is applied at this stage
		
		\item Its role is to pass the data to the first hidden layer
	\end{itemize}
	
\end{frame}


\subsection{Hidden layers}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item Each neuron computes a weighted sum followed by a nonlinear activation:
		\[
		z^{(l)} = W^{(l)} a^{(l-1)} + b^{(l)}, 
		\quad a^{(l)} = f(z^{(l)})
		\]
		
		\item Each layer transforms the representation of the data
		
		\item Stacking multiple layers allows learning complex patterns
		
		\item These layers extract intermediate (hidden) features
		
		\item Deeper layers capture more abstract representations
	\end{itemize}
	
\end{frame}


\subsection{Output layer}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item The output layer produces the final prediction:
		\[
		\hat{y} = a^{(L)}
		\]
		
		\item Its structure depends on the task:
		\begin{itemize}
			\item Regression: linear activation
			\item Binary classification: sigmoid activation
			\item Multiclass classification: softmax activation
		\end{itemize}
		
		\item The output is compared to the true target using a cost function
	\end{itemize}
	
\end{frame}


\section{Training}

\begin{frame}
	\frametitle{\insertshortsubtitle}
	\framesubtitle{\insertsection}
	
	\begin{itemize}
		\item Training a neural network consists in learning the parameters (weights and biases)
		
		\item Three main components are involved:
		\begin{itemize}
			\item \optword{Cost function}: measures the error between predictions and true targets
			
			\item \optword{Backpropagation}: computes the gradients of the cost function
			
			\item \optword{Optimization algorithm}: updates the parameters to minimize the cost
		\end{itemize}
		
		\item These steps are repeated iteratively until convergence
	\end{itemize}
	
\end{frame}


\subsection{Cost functions}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item A \optword{loss function} measures the error for a single prediction:
		\[
		\mathcal{L}(y, \hat{y})
		\]
		
		\item A \optword{cost function} (or objective) is the average loss over the dataset:
		\[
		J(\theta) = \frac{1}{N} \sum_{i=1}^{N} \mathcal{L}(y^{(i)}, \hat{y}^{(i)})
		\]
		
		\item The goal of training is to find parameters $\theta$ that minimize the cost:
		\[
		\theta^* = \arg\min_{\theta} J(\theta)
		\]
		
		\item The choice of the loss function depends on the task (regression, classification).
	\end{itemize}
	
\end{frame}

		\]

	\begin{itemize}
		\item \optword{Advantages}
		\begin{itemize}
			\item Combines the stability of MSE for small errors and robustness of MAE for large errors.
			\item Behaves like MSE near zero error and like MAE for large errors.

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Classification (BCE)}

	\begin{block}{Binary Cross Entropy loss}
		\[\mathcal{L}(y, \hat{y}) = - (y \log(\hat{y}) + (1 - y) \log(1 - \hat{y}))\]
		
		\[
		\frac{\partial \mathcal{L}}{\partial \hat{y}} = \frac{\hat{y} - y}{\hat{y} - \hat{y}^2}
		\]
	\end{block}
	
	\begin{itemize}
		\item \optword{Advantages}
		\begin{itemize}
			\item Matches naturally with sigmoid output interpreted as probabilities.
			\item Provides better-calibrated probability estimates.
		\end{itemize}
		\item \optword{Disadvantages}
		\begin{itemize}
			\item Focuses on likelihood maximization rather than margin maximization.
		\end{itemize}
	\end{itemize}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Classification (Hinge)}

	\begin{block}{Hinge loss (SVM loss)}
		\[\mathcal{L}(y, \hat{y}) = \max(0, 1 - y \hat{y}), \quad y \in \{-1, 1\}\]
		
		\[
		\frac{\partial \mathcal{L}}{\partial \hat{y}} =
		\begin{cases}
			0 & \text{if } y \hat{y} > 1 \\
			-y & \text{if } y \hat{y} < 1 \\
			\text{undefined (subgradient)} & \text{if } y \hat{y} = 1
		\end{cases}
		\]
	\end{block}
	
	\begin{itemize}
		\item \optword{Advantages}
		\begin{itemize}
			\item Encourages large-margin separation between classes.
		\end{itemize}
		\item \optword{Disadvantages}
		\begin{itemize}
			\item Not smooth at the margin, requiring subgradient-based optimization.
			\item Does not produce probabilistic outputs.
		\end{itemize}
	\end{itemize}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Multinomial Classification (Cross entropy)}

	\begin{block}{Cross Entropy loss}
		\[\mathcal{L}(y, \hat{y}) = - \sum\limits_{k=1}^{K} y_{k} \log(\hat{y}_{k}) \text{, where } K \text{ is the number of classes} \]
		
		\[
		\frac{\partial \mathcal{L}}{\partial \hat{y}_k} = - \frac{y_{k}}{\hat{y}_{k}}
		\]
	\end{block}
	
	\begin{itemize}
		\item \optword{Note}
		\begin{itemize}
			\item In neural networks, $\hat{y}$ is typically obtained using the softmax function.
		\end{itemize}
		
		\item \optword{Advantages}
		\begin{itemize}
			\item Well suited for probabilistic multi-class classification.
			\item Works naturally with softmax activation.
		\end{itemize}
		
		\item \optword{Disadvantages}
		\begin{itemize}
			\item Sensitive to confident wrong predictions.
		\end{itemize}
	\end{itemize}

\end{frame}

\subsection{Optimization algorithms}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection}
	
	\begin{itemize}
		\item The goal of training is to minimize the cost function:
		\[
		\theta^* = \arg\min_{\theta} J(\theta)
		\]
		
		\item \optword{Optimization algorithms} define how parameters are updated to reduce the cost.
		
		\item They iteratively adjust the parameters using gradients of the cost function.
		
		\item Different strategies exist depending on:
		\begin{itemize}
			\item computational efficiency;
			\item dataset size;
			\item convergence stability.
		\end{itemize}
		
		\item Common methods include Gradient Descent, SGD, and adaptive optimizers (Adam, RMSProp).
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Gradient descent}

	\begin{figure}
		\vgraphpage[.7\textheight]{steepest.png}
		\caption{Gradient descent illustration \cite{2020-calin}}
	\end{figure}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Gradient descent}

	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T $}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		\While{$ t < T$ and no convergence}{
			$\hat{Y} \leftarrow f_\theta(X)$\;
			$\theta \leftarrow \theta - \alpha \nabla_\theta J(Y, \hat{Y}; \theta)$\;
			$t \leftarrow t + 1 $\;
		}
		\caption{Gradient descent}
	\end{algorithm}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Gradient descent}

	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Iteratively updates parameters in the direction of the negative gradient of the loss.
			\item Convergence rate depends on the convexity and smoothness of the objective.
		\end{itemize}
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Guarantees convergence to a global optimum for convex objectives
		\end{itemize}
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item Requires computing gradients over the full dataset at each update.
			\item Computationally expensive for large datasets.
			\item Can converge to local minima for non-convex objectives.
		\end{itemize}
	\end{itemize}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Stochastic gradient descent}

	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T$}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		\While{$ t < T$ and no convergence}{
			Randomly sample $(x, y) = (x^{(i)}, y^{(i)})$ from $(X, Y)$\;
			$\hat{y} \leftarrow f_\theta(x)$\;
			$\theta \leftarrow \theta - \alpha \nabla_\theta \mathcal{L}(y, \hat{y}; \theta)$\;
			$t \leftarrow t + 1 $\;
		}
		\caption{Stochastic gradient descent}
	\end{algorithm}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Stochastic gradient descent}

	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Performs one parameter update per training sample.
		\end{itemize}
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Stochasticity helps escape shallow local minima and saddle points.
		\end{itemize}
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item Convergence is noisy and less stable due to high variance in updates.
		\end{itemize}	
	\end{itemize}

\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Mini-Batch gradient descent}

	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T, b $}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		\While{$ t < T$ and no convergence}{
			Randomly mix data\;
			Divide data to $b$ batches\;
			\ForEach{mini-batch $ (X^b, Y^b) \in $ batches}{
				$\hat{Y}^b \leftarrow f_\theta(X^b)$\;
				$ \theta = \theta - \alpha \nabla_\theta J(Y^b, \hat{Y}^b; \theta) $\;
			}
	%		\tcp{Each mini-batch produces one parameter update}
			$t \leftarrow t + 1 $\;
		}
		\caption{Mini-Batch gradient descent}
	\end{algorithm}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Mini-Batch gradient descent}

	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Training is done on batches (subsets) of the dataset.
			\item Uses a hyper-parameter $b$ defining the batch size.
		\end{itemize}
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Frequent updates introduce noise that can improve exploration of the loss landscape.
			\item Efficient with large training data.
		\end{itemize}
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item Requires tuning of batch size $b$.
			\item Performance depends on batch size selection.
		\end{itemize}
	\end{itemize}

\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Adaptive gradient descent (AdaGrad)}

	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T$}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		Initialize $v$ ($|v| = |\theta|$) to zero\;
		\While{$ t < T$ and no convergence}{
			$\hat{Y} \leftarrow f_\theta(X)$\;
			$ g \leftarrow \nabla_\theta J(Y, \hat{Y}; \theta)$\;
			$ v = v +\, g \odot g $\;
			$ \theta = \theta - \frac{\alpha}{\sqrt{v} + \epsilon} \odot g $\;
			\tcp{$\epsilon \approx 10^{-8}$ to avoid division by zero}
			$t \leftarrow t + 1 $\;
		}
		\caption{AdaGrad \cite{2011-duchi-al}}
	\end{algorithm}

\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Adaptive Gradient Descent (AdaGrad)}
	
	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Adapts the learning rate for each parameter individually.
			\item Uses the accumulated sum of squared past gradients.
		\end{itemize}
		
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Automatically reduces the learning rate for frequently updated parameters.
			\item Works well for sparse features.
		\end{itemize}
		
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item The learning rate continuously decreases over time and may become too small.
			\item Not well-suited for non-convex optimization in deep networks.
		\end{itemize}
	\end{itemize}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Root Mean Square Propagation (RMSProp)}
	
	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T, \beta$}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		Initialize $ v $ ($|v| = |\theta|$) to zero\;
		\While{$ t < T$ and no convergence}{
			$\hat{Y} \leftarrow f_\theta(X)$\;
			$ g \leftarrow \nabla_\theta J(Y, \hat{Y}; \theta)$\;
			$ v \leftarrow \beta v + (1-\beta)\, g \odot g $\;
			$ \theta \leftarrow \theta - \frac{\alpha}{\sqrt{v} + \epsilon} \odot g $\;
			$t \leftarrow t + 1 $\;
		}
		\caption{RMSProp}
	\end{algorithm}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Root Mean Square Propagation (RMSProp)}
	
	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Uses an exponential moving average of squared gradients.
			\item Normalizes gradient updates using recent magnitude statistics.
			\item Typical value: $\beta \approx 0.9$
		\end{itemize}
		
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Fixes AdaGrad's rapid learning-rate decay.
			\item Works well for non-stationary and non-convex optimization.
		\end{itemize}
		
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item Can lead to oscillations near minima depending on hyperparameters.
			\item Sensitive to the choice of learning rate and decay factor.
		\end{itemize}
	\end{itemize}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Adaptive Moment Estimation (Adam)}
	
	\begin{algorithm}[H]
		\KwData{$ X, Y, \alpha, T, \beta_1, \beta_2$}
		\KwResult{$ \theta $}
		Initialize $ \theta $ ; $ t = 0 $\;
		Initialize $ m $ and $ v $ ($|m| = |v| = |\theta|$) to zero\;
		\While{$ t < T$ and no convergence}{
			$\hat{Y} \leftarrow f_\theta(X)$\;
			$ g \leftarrow \nabla_\theta J(Y, \hat{Y}; \theta)$\;
			$ m \leftarrow \beta_1 m + (1-\beta_1) g; \quad \hat{m} \leftarrow \frac{m}{1-\beta_1^t} $\;
			$ v \leftarrow \beta_2 v + (1-\beta_2)\, g \odot g; \quad \hat{v} \leftarrow \frac{v}{1-\beta_2^t} $\;
			$ \theta \leftarrow \theta - \frac{\alpha \hat{m}}{\sqrt{\hat{v}} + \epsilon} $\;
			$t \leftarrow t + 1 $\;
		}
		\caption{Adam \cite{2015-kingma-ba}}
	\end{algorithm}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Adaptive Moment Estimation (Adam)}
	
	\begin{itemize}
		\item \optword{Properties}: 
		\begin{itemize}
			\item Combines first moment (mean of gradients) and second moment (uncentered variance).
			\item Uses bias correction for both estimates.
			\item Typical values: $\beta_1 = 0.9$, $\beta_2 = 0.999$
		\end{itemize}
		
		\item \optword{Advantages}: 
		\begin{itemize}
			\item Fast and stable convergence in practice.
			\item Works well for large-scale and high-dimensional non-convex problems.
		\end{itemize}
		
		\item \optword{Disadvantages}: 
		\begin{itemize}
			\item May converge to suboptimal solutions in some settings.
		\end{itemize}
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Example - Output layer (chain rule)}
	
	\begin{exampleblock}{Sigmoid + BCE (separate functions)}
		\centering
		$\hat{y} = \sigma(z), \quad
		J = - (y \log(\hat{y}) + (1-y)\log(1-\hat{y}))$

		$
		\frac{\partial J}{\partial z} = \delta_{1}^{(4)} =
		\underbrace{\frac{\partial J}{\partial \hat{y}}}_{\text{Loss}}
		\cdot
		\underbrace{\frac{\partial \hat{y}}{\partial z}}_{\text{Activation}}
		= 
		(\frac{\hat{y} - y}{\hat{y} - \hat{y}^2}) \cdot (\sigma(z) (1- \sigma(z)))
		$
	\end{exampleblock}
	
	\begin{itemize}
		\item Chain rule is applied across all components of the model
		\item This corresponds to the computational graph (e.g., PyTorch)
	\end{itemize}
	
	
	$ 
	\frac{\partial J}{\partial f_{1}^{(4)}} = \frac{(0.840, 0.843) - (0, 1)}{(0.840, 0.843) - (0.840, 0.843)^2} 
	= (6.25, -1.186)
	$
	
	$ 
	\frac{\partial f_{1}^{(4)}}{\partial z_{1}^{(4)}} = (0.840, 0.843) (0.160, 0.157) = (0.134, 0.132)
	$
	
	$
	\delta_{1}^{(4)} = (6.25, -1.186) \odot (0.134, 0.132) \approx (0.838, -0.157)
	$
	
	
\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Example - Output layer (simplification)}
	
	\begin{exampleblock}{Result of the chain rule}

		$\hat{y} = \sigma(z), \quad
		J = - (y \log(\hat{y}) + (1-y)\log(1-\hat{y}))$
		
		$\frac{\partial J}{\partial z} = \delta_{1}^{(4)} = \hat{y} - y$
	\end{exampleblock}
	
	\begin{itemize}
		\item Much simpler expression for backpropagation
		\item Interpreted as the \optword{prediction error}
		\item Widely used in practice (e.g., BCE with logits)
	\end{itemize}
	
	$
	\delta_{1}^{(4)} = (0.840, 0.843) - (0, 1) = (0.840, -0.157)
	$
	
	Both approaches give the same result (up to rounding)

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: General case}
	
	\begin{itemize}
		\item Calculate $\delta^{(l)}$ where $l$ is the layer's number
	\end{itemize}
	
	\[ 
	\delta^{(out)} = 
	\frac{\partial J}{\partial f^{(out)}} \odot \frac{\partial f^{(out)}}{\partial z^{(out)}}
	\,,\,
	\delta^{(l)} = \left(W^{(l+1)}\right)^T \delta^{(l+1)} \odot \frac{\partial f^{(l)}}{\partial z^{(l)}}
	\]
	
	\begin{itemize}
		\item Calculate gradients
	\end{itemize}
	
	\[ 
	\frac{\partial J}{\partial W^{(l)}} = \frac{1}{m} a^{(l-1)} (\delta^{(l)})^T
	\,,\,
	\frac{\partial J}{\partial b^{(l)}} = \frac{1}{m} \sum \delta^{(l)}
	\]
	
	\begin{itemize}
		\item Update parameters
	\end{itemize}
	
	\[ 
	W^{(l)} = W^{(l)} - \alpha \frac{\partial J}{\partial W^{(l)}}
	\,,\,
	b^{(l)} = b^{(l)} - \alpha \frac{\partial J}{\partial b^{(l)}}
	\]
	
	
\end{frame}

\subsection{Regularization}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Motivation}
	
	\begin{itemize}
		\item Neural networks can easily \optword{overfit} the training data.
		\item Overfitting: good performance on training data, poor on unseen data.
		\item \optword{Regularization} aims to improve \keyword{generalization}.
	\end{itemize}
	
	\begin{block}{Idea}
		Reduce model complexity or constrain learning to avoid fitting noise.
	\end{block}
	
	\begin{itemize}
		\item Trade-off: \optword{bias vs variance}
	\end{itemize}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Common techniques}
	
	\begin{itemize}
		\item \optword{Weight regularization}
		\begin{itemize}
			\item L2 (Ridge): penalizes large weights
			\item L1 (Lasso): promotes sparsity
		\end{itemize}
		
		\item \optword{Dropout}
		\item \optword{Early stopping}
		\item \optword{Data augmentation}
	\end{itemize}
	
	\begin{block}{Key idea}
		Prevent the model from relying too much on specific features.
	\end{block}
	
\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: L2 regularization}
	
	\begin{block}{Regularized loss}
		\[
		J_{reg} = J + \lambda \sum w^2
		\]
	\end{block}
	
	\begin{itemize}
		\item Penalizes large weights
		\item Encourages smoother models
		\item $\lambda$: regularization strength
	\end{itemize}
	
	\begin{block}{Effect on update}
		Weights are slightly pushed toward zero at each step
	\end{block}
	
\end{frame}

\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Dropout \cite{2014-srivastava}}
	
	\begin{itemize}
		\item Randomly deactivate neurons during training.
		\item Each neuron is kept with probability $p$ (keep rate).
		\item Equivalent to sampling a different sub-network at each iteration.
	\end{itemize}
	
	\begin{block}{Mechanism}
		\[
		a^{(l)} \leftarrow \frac{a^{(l)} \odot m}{p}, \quad m \sim \text{Bernoulli}(p)
		\]
	\end{block}
	
	\begin{itemize}
		\item Applied only during training;
		\item Ensures same expected activation at training and inference:
		\[
		\mathbb{E}[a_{\text{train}}] = \mathbb{E}[a_{\text{test}}]
		\]
		\item At inference: no scaling is applied.
	\end{itemize}
	
\end{frame}


\begin{frame}
	\frametitle{\insertshortsubtitle: \insertsection}
	\framesubtitle{\insertsubsection: Dropout effects}
	
	\begin{itemize}
		\item \optword{Advantages}
		\begin{itemize}
			\item Reduces overfitting
			\item Acts like an ensemble of models
		\end{itemize}
		
		\item \optword{Disadvantages}
		\begin{itemize}
			\item Slower convergence
			\item Requires tuning of dropout rate
		\end{itemize}
	\end{itemize}
	
\end{frame}
